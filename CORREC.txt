--- INSTALLER (une fois par machine) ------------------------------------------

git clone https://github.com/Thomasrbm/Cloud-1.git
cd Cloud-1
sudo apt update
sudo apt install -y ansible
ansible-galaxy collection install -r requirements.yml

# group_vars/webservers.yml n'est PAS dans git (secrets).
# Le copier depuis une machine qui l'a (cle USB, scp) :
cp /media/usb/webservers.yml group_vars/webservers.yml
head -1 group_vars/webservers.yml        # doit afficher $ANSIBLE_VAULT;1.1;AES256


--- DEPLOYER -------------------------------------------------------------------

ansible -m ping webservers
ansible-playbook playbook.yml --ask-vault-pass


--- VERIFIER -------------------------------------------------------------------

--- SUPPRIMER ------------------------------------------------------------------

ansible-playbook teardown.yml                          # stack + donnees
ansible-playbook teardown.yml -e full_wipe=true        # + Docker, ufw : Ubuntu nu


--- DEMO CORRECTEUR (reset + redeploiement complet) ----------------------------

--- AJOUTER UNE MACHINE --------------------------------------------------------

cat ~/.ssh/id_ed25519.pub

ansible-playbook playbook.yml --ask-vault-pass
